    <!-- Web Audio 音效系统 -->
    <script>
        class SoundFX {
            constructor() {
                this.ctx = null;
                this.muted = false;
            }

            init() {
                if (!this.ctx) {
                    this.ctx = new (window.AudioContext || window.webkitAudioContext)();
                }
            }

            playClick() {
                if (this.muted) return;
                this.init();
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(600, this.ctx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(300, this.ctx.currentTime + 0.05);
                gain.gain.setValueAtTime(0.15, this.ctx.currentTime);
                gain.gain.linearRampToValueAtTime(0.01, this.ctx.currentTime + 0.05);
                osc.connect(gain);
                gain.connect(this.ctx.destination);
                osc.start();
                osc.stop(this.ctx.currentTime + 0.05);
            }

            playTypewriter() {
                if (this.muted) return;
                this.init();
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(750 + Math.random() * 200, this.ctx.currentTime);
                gain.gain.setValueAtTime(0.02, this.ctx.currentTime);
                gain.gain.linearRampToValueAtTime(0.001, this.ctx.currentTime + 0.03);
                osc.connect(gain);
                gain.connect(this.ctx.destination);
                osc.start();
                osc.stop(this.ctx.currentTime + 0.03);
            }

            playChime(isGood = true) {
                if (this.muted) return;
                this.init();
                const freqs = isGood ? [523.25, 659.25, 783.99, 1046.50] : [300, 260, 220, 180];
                freqs.forEach((f, idx) => {
                    const osc = this.ctx.createOscillator();
                    const gain = this.ctx.createGain();
                    osc.type = isGood ? 'sine' : 'sawtooth';
                    osc.frequency.setValueAtTime(f, this.ctx.currentTime + idx * 0.08);
                    gain.gain.setValueAtTime(0.1, this.ctx.currentTime + idx * 0.08);
                    gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + idx * 0.08 + 0.4);
                    osc.connect(gain);
                    gain.connect(this.ctx.destination);
                    osc.start(this.ctx.currentTime + idx * 0.08);
                    osc.stop(this.ctx.currentTime + idx * 0.08 + 0.4);
                });
            }
        }
        const sfx = new SoundFX();
    </script>

        window.addEventListener("DOMContentLoaded", () => {

        // 渲染剧情节点核心引擎
        function renderNode(nodeId, pushHistory = true) {
            const node = NODES[nodeId];
            if (!node) return;

            if (pushHistory && currentNodeId !== nodeId) {
                historyStack.push({
                    nodeId: currentNodeId,
                    flags: new Set(userFlags)
                });
            }

            currentNodeId = nodeId;
            updateUndoButton();

            // 切换背景图片
            const bgEl = document.getElementById("bg-container");
            bgEl.style.backgroundImage = `url('${node.bg}')`;

            // 更新头部与困境标签
            document.getElementById("node-tag").innerHTML = `节点: <span class="text-slate-200 font-mono">${nodeId}</span>`;
            document.getElementById("dilemma-text").innerText = node.dilemma || "抉择";
            document.getElementById("speaker-label").innerText = node.speaker || "旁白";
            document.getElementById("scene-location").innerHTML = `<i class="fa-solid fa-location-dot"></i> ${node.location || '场景'}`;

            // 控制知识卡入口按钮
            const cardBtn = document.getElementById("card-btn");
            if (node.knowledgeCard) {
                cardBtn.classList.remove("hidden");
            } else {
                cardBtn.classList.add("hidden");
            }

            // 逐字打字机渲染
            const textEl = document.getElementById("narrative-text");
            textEl.innerHTML = "";
            textEl.classList.add("typewriter-cursor");

            if (currentTypewriterTimer) clearInterval(currentTypewriterTimer);

            let charIndex = 0;
            const fullText = node.text;

            sfx.playClick();

            currentTypewriterTimer = setInterval(() => {
                if (charIndex < fullText.length) {
                    if (fullText[charIndex] === '<') {
                        const closeTagIndex = fullText.indexOf('>', charIndex);
                        if (closeTagIndex !== -1) {
                            charIndex = closeTagIndex + 1;
                        }
                    } else {
                        charIndex++;
                    }
                    textEl.innerHTML = fullText.substring(0, charIndex);
                    if (charIndex % 3 === 0) sfx.playTypewriter();
                } else {
                    clearInterval(currentTypewriterTimer);
                    textEl.classList.remove("typewriter-cursor");
                }
            }, 18);

            // 动态生成选项按钮
            const choicesContainer = document.getElementById("choices-container");
            choicesContainer.innerHTML = "";

            let choicesList = node.choices || [];

            // 对 'expose' 节点根据前面积累的标志位追加条件分支结局
            if (node.dynamicChoices) {
                choicesList = [
                    { text: "A. 正视全部问题，发布诚恳长文道歉，同时发起「媒体如何报道心理疾病」的行业讨论", ending: "救赎之路" },

                ];

                if (userFlags.has("bought_medical_record") || userFlags.has("published_illegal_record")) {
                    choicesList.push({
                        text: "E. 曾参与购买/传播非法病历并且拒绝认错（触犯法律）",
                        ending: "高墙之内"
                    });
                }

                if (userFlags.has("covered_up_black_market")) {
                    choicesList.push({
                        text: "F. 明知病历黑色链条存在却选择帮忙掩盖",
                        ending: "共犯"
                    });
                }
            }

            choicesList.forEach((choice) => {
                const btn = document.createElement("button");
                btn.className = "vn-button w-full text-left p-3.5 sm:p-4 rounded-xl text-xs sm:text-sm text-slate-200 flex items-center justify-between group shadow-sm";
                btn.innerHTML = `
                    <span class="pr-2 leading-relaxed">${choice.text}</span>
                    <i class="fa-solid fa-chevron-right text-slate-500 group-hover:text-blue-400 group-hover:translate-x-1 transition-all text-xs flex-shrink-0"></i>
                `;

                btn.onclick = () => {
                    sfx.playClick();
                    if (choice.flag) {
                        userFlags.add(choice.flag);
                    }

                    if (choice.ending) {
                        triggerEnding(choice.ending);
                    } else if (choice.next) {
                        renderNode(choice.next);
                    }
                };

                choicesContainer.appendChild(btn);
            });
        }

        // 撤销/退回上一节点
        function undoChoice() {
            if (historyStack.length === 0) return;
            sfx.playClick();
            const lastState = historyStack.pop();
            userFlags = lastState.flags;
            renderNode(lastState.nodeId, false);
        }

        function updateUndoButton() {
            const undoBtn = document.getElementById("undo-btn");
            if (historyStack.length > 0) {
                undoBtn.removeAttribute("disabled");
            } else {
                undoBtn.setAttribute("disabled", "true");
            }
        }

        // 触发结局
        function triggerEnding(endingKey) {
            const ending = ENDINGS[endingKey];
            if (!ending) return;

            unlockedEndings.add(endingKey);
            localStorage.setItem("shutter_endings_v2", JSON.stringify(Array.from(unlockedEndings)));
            updateUnlockedCount();

            sfx.playChime(ending.type === "good");

            const badgeEl = document.getElementById("ending-badge");
            if (ending.type === "good") {
                badgeEl.className = "mb-3 px-3 py-1 rounded-full text-xs font-bold tracking-widest uppercase border bg-emerald-500/20 text-emerald-400 border-emerald-500/40";
                badgeEl.innerText = "好结局";
            } else if (ending.type === "neutral") {
                badgeEl.className = "mb-3 px-3 py-1 rounded-full text-xs font-bold tracking-widest uppercase border bg-amber-500/20 text-amber-400 border-amber-500/40";
                badgeEl.innerText = "中性结局";
            } else {
                badgeEl.className = "mb-3 px-3 py-1 rounded-full text-xs font-bold tracking-widest uppercase border bg-red-500/20 text-red-400 border-red-500/40";
                badgeEl.innerText = "坏结局";
            }

            document.getElementById("ending-icon").innerText = ending.icon;
            document.getElementById("ending-title").innerText = ending.title;
            document.getElementById("ending-desc").innerText = ending.desc;
            document.getElementById("ending-quote").innerText = ending.quote;

            const modal = document.getElementById("ending-modal");
            modal.classList.remove("hidden");
            setTimeout(() => modal.classList.remove("opacity-0"), 10);
        }

        function closeEndingModal() {
            const modal = document.getElementById("ending-modal");
            modal.classList.add("opacity-0");
            setTimeout(() => modal.classList.add("hidden"), 300);
        }

        function restartGame() {
            userFlags.clear();
            historyStack = [];
            renderNode("start", false);
        }

        function openKnowledgeModal() {
            const node = NODES[currentNodeId];
            if (!node || !node.knowledgeCard) return;

            sfx.playClick();
            document.getElementById("knowledge-card-body").innerHTML = node.knowledgeCard;

            const modal = document.getElementById("knowledge-modal");
            modal.classList.remove("hidden");
            setTimeout(() => modal.classList.remove("opacity-0"), 10);
        }

        function closeKnowledgeModal() {
            const modal = document.getElementById("knowledge-modal");
            modal.classList.add("opacity-0");
            setTimeout(() => modal.classList.add("hidden"), 300);
        }

        function toggleGalleryModal() {
            sfx.playClick();
            const modal = document.getElementById("gallery-modal");

            if (modal.classList.contains("hidden")) {
                renderGalleryGrid();
                modal.classList.remove("hidden");
                setTimeout(() => modal.classList.remove("opacity-0"), 10);
            } else {
                modal.classList.add("opacity-0");
                setTimeout(() => modal.classList.add("hidden"), 300);
            }
        }

        function renderGalleryGrid() {
            const grid = document.getElementById("gallery-grid");
            grid.innerHTML = "";

            Object.keys(ENDINGS).forEach((key) => {
                const ending = ENDINGS[key];
                const isUnlocked = unlockedEndings.has(key);

                const card = document.createElement("div");
                card.className = `p-3.5 rounded-xl border transition flex flex-col justify-between ${
                    isUnlocked
                        ? "bg-slate-800/80 border-slate-700/80 text-slate-200"
                        : "bg-slate-950/60 border-slate-800/60 text-slate-600 opacity-60"
                }`;

                card.innerHTML = `
                    <div>
                        <div class="flex items-center justify-between mb-1.5">
                            <span class="text-xl">${isUnlocked ? ending.icon : '🔒'}</span>
                            <span class="text-[10px] font-bold px-2 py-0.5 rounded ${
                                isUnlocked 
                                    ? (ending.type === 'good' ? 'bg-emerald-950 text-emerald-400 border border-emerald-800' : ending.type === 'neutral' ? 'bg-amber-950 text-amber-400 border border-amber-800' : 'bg-red-950 text-red-400 border border-red-800')
                                    : 'bg-slate-900 text-slate-600'
                            }">
                                ${isUnlocked ? (ending.type === 'good' ? '好结局' : ending.type === 'neutral' ? '中性结局' : '坏结局') : '未解锁'}
                            </span>
                        </div>
                        <h4 class="font-bold text-sm text-slate-200 ${!isUnlocked && 'blur-[2px]'}">${ending.title}</h4>
                        <p class="text-xs mt-1 text-slate-400 leading-snug line-clamp-3">${isUnlocked ? ending.desc : '探索分支解锁此结局。'}</p>
                    </div>
                `;

                grid.appendChild(card);
            });

            document.getElementById("gallery-progress-text").innerText = unlockedEndings.size;
        }

        function updateUnlockedCount() {
            document.getElementById("unlocked-count").innerText = unlockedEndings.size;
        }

        function toggleMute() {
            sfx.muted = !sfx.muted;
            const btn = document.getElementById("sound-btn");
            btn.innerHTML = sfx.muted ? '<i class="fa-solid fa-volume-xmark text-red-400"></i>' : '<i class="fa-solid fa-volume-high"></i>';
        }
